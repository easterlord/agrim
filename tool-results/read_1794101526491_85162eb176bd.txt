   156→
   157→export function CategoryBar({
   158→  platforms,
   159→  active,
   160→  onSelect,
   161→}: {
   162→  platforms: Platform[];
   163→  active: string | null;
   164→  onSelect: (id: string) => void;
   165→}) {
   166→  const { t } = useI18n();
   167→  const [kind, setKind] = useState<"social" | "game">("social");
   168→
   169→  const filtered = useMemo(() => platforms.filter((p) => p.kind === kind), [platforms, kind]);
   170→
   171→  // derived effective selection — no state-sync effects needed
   172→  const activeId = active && filtered.some((p) => p.id === active) ? active : filtered[0]?.id ?? null;
   173→
   174→  const switchKind = (k: "social" | "game") => {
   175→    setKind(k);
   176→    const list = platforms.filter((p) => p.kind === k);
   177→    if (list.length && !list.some((p) => p.id === active)) onSelect(list[0].id);
   178→  };
   179→
   180→  return (
   181→    <div className="space-y-4">
   182→      <div className="flex justify-center">
   183→        <Segmented
   184→          value={kind}
   185→          onChange={(v) => switchKind(v as "social" | "game")}
   186→          options={[
   187→            { value: "social", label: <span className="flex items-center gap-1.5"><Sparkles size={12} /> {t("shop.social")}</span> },
   188→            { value: "game", label: <span className="flex items-center gap-1.5"><Gamepad2 size={12} /> {t("shop.games")}</span> },
   189→          ]}
   190→        />
   191→      </div>
   192→
   193→      <div className="no-scrollbar -mx-4 flex gap-2 overflow-x-auto px-4 pb-1 sm:mx-0 sm:flex-wrap sm:justify-center sm:px-0">
   194→        <AnimatePresence mode="popLayout">
   195→          {filtered.map((p, i) => {
   196→            const isActive = p.id === activeId;
   197→            return (
   198→              <motion.button
   199→                layout
   200→                key={p.id}
   201→                initial={{ opacity: 0, scale: 0.85, y: 10 }}
   202→                animate={{ opacity: 1, scale: 1, y: 0 }}
   203→                exit={{ opacity: 0, scale: 0.9 }}
   204→                transition={{ delay: i * 0.03, type: "spring", stiffness: 380, damping: 28 }}
   205→                whileHover={{ y: -2 }}
   206→                whileTap={{ scale: 0.95 }}
   207→                onClick={() => onSelect(p.id)}
   208→                aria-pressed={isActive}
   209→                className={cn(
   210→                  "glass flex shrink-0 items-center gap-2 rounded-full px-4 py-2.5 text-sm font-semibold transition-all",
   211→                  isActive && "pill-active",
   212→                )}
   213→                style={isActive ? { color: p.color } : undefined}
   214→              >
   215→                <PlatformIcon id={p.id} size={17} className={cn(!isActive && "text-muted-foreground")} />
   216→                <span className={cn(!isActive && "text-muted-foreground")}>{p.name}</span>
   217→              </motion.button>
   218→            );
   219→          })}
   220→        </AnimatePresence>
   221→      </div>
   222→    </div>
   223→  );
   224→}

   263→export function Configurator({
   264→  platformId,
   265→  onSelectPlatform,
   266→}: {
   267→  platformId: string | null;
   268→  onSelectPlatform: (id: string) => void;
   269→}) {
   270→  const { t } = useI18n();
   271→  const { platforms } = useCatalog();
   272→  const { settings } = useSettings();
   273→  const { currency } = useUI();
   274→  const draft = useDraft();
   275→
   276→  const [typeId, setTypeId] = useState<string | null>(draft.typeId);
   277→  const [qty, setQty] = useState<number>(draft.quantity ?? 1000);
   278→  const [packageId, setPackageId] = useState<string | null>(draft.packageId);
   279→  const [playerId, setPlayerId] = useState(draft.playerId ?? "");
   280→  const [link, setLink] = useState(draft.link ?? "");
   281→  const [err, setErr] = useState("");
   282→
   283→  /* derived selection chain — zero sync effects */
   284→  const platform = platforms.find((p) => p.id === platformId) ?? platforms[0] ?? null;
   285→  const activeServices = useMemo(
   286→    () => platform?.services.filter((s) => s.active) ?? [],
   287→    [platform],
   288→  );
   289→  const service = activeServices.find((s) => s.id === typeId) ?? activeServices[0] ?? null;
   290→  const isGame = platform?.kind === "game";
   291→  const effPackageId =
   292→    service?.packages?.length
   293→      ? service.packages.some((p) => p.id === packageId)
   294→        ? packageId
   295→        : service.packages[0].id
   296→      : null;
   297→
   298→  const price = useMemo(() => {
   299→    if (!service) return { usd: 0, valid: false, qty: 0 };
   300→    if (service.packages?.length) {
   301→      const pk = findPackage(service, effPackageId);
   302→      return { usd: pk?.priceUSD ?? 0, valid: true, qty: pk?.qty ?? 0 };
   303→    }
   304→    const r = calcQtyPrice(service, qty);
   305→    return { usd: r.usd, valid: r.valid, qty: r.qty };
   306→  }, [service, effPackageId, qty]);
   307→
   308→  const shown = currency === "NPR" ? fmtNPR(convertUSD(price.usd, settings.exchangeRate)) : fmtUSD(price.usd);
   309→
   310→  const buy = () => {
   311→    setErr("");
   312→    if (!platform || !service) return;
   313→    if (service.packages?.length && !effPackageId) {
   314→      setErr(t("shop.package"));
   315→      return;
   316→    }
   317→    if (!service.packages?.length && !price.valid) {
   318→      setErr(`${t("shop.min")}: ${service.minQty ?? 100}`);
   319→      return;
   320→    }
   321→    if (isGame && !playerId.trim()) {
   322→      setErr(t("shop.uidHint"));
   323→      document.getElementById("cfg-player")?.focus();
   324→      return;
   325→    }
   326→    draft.setDraft({
   327→      platformId: platform.id,
   328→      typeId: service.id,
   329→      quantity: service.packages?.length ? null : price.qty,
   330→      packageId: service.packages?.length ? effPackageId : null,
   331→      playerId: playerId.trim(),
   332→      link: link.trim(),
   333→    });
   334→    navigate("/checkout");
   335→  };
   336→
   337→  const sliderFill =
   338→    service && service.maxQty && service.maxQty > (service.minQty ?? 100)
   339→      ? `${(((qty - (service.minQty ?? 100)) / (service.maxQty - (service.minQty ?? 100))) * 100).toFixed(1)}%`
   340→      : "50%";
   341→
   342→  return (
   343→    <GlassCard className="overflow-hidden p-0">
   344→      <div className="grid lg:grid-cols-[320px_1fr]">
   345→        {/* service list */}
   346→        <div className="border-b border-border/60 p-5 lg:border-b-0 lg:border-r">
   347→          {platform ? (
   348→            <div className="mb-4 flex items-center gap-3">
   349→              <span className="glass flex h-11 w-11 items-center justify-center rounded-xl" style={{ color: platform.color }}>
   350→                <PlatformIcon id={platform.id} size={22} />
   351→              </span>
   352→              <div className="min-w-0">
   353→                <p className="truncate text-sm font-bold">{platform.name}</p>
   354→                <p className="text-[11px] text-muted-foreground">
   355→                  {activeServices.length} {t("nav.services").toLowerCase()}
   356→                </p>
   357→              </div>
   358→            </div>
   359→          ) : null}
   360→
   361→          <div className="max-h-72 space-y-1.5 overflow-y-auto pr-1 lg:max-h-[420px]">
   362→            {activeServices.map((s) => (
   363→              <ServiceRow key={s.id} service={s} active={s.id === service?.id} onClick={() => setTypeId(s.id)} />
   364→            ))}
   365→            {platform && activeServices.length === 0 ? (
   366→              <p className="py-8 text-center text-xs text-muted-foreground">{t("shop.selectPrompt")}</p>
   367→            ) : null}
   368→          </div>
   369→        </div>
   370→
   371→        {/* config panel */}
   372→        <div className="p-5 sm:p-6">
   373→          <AnimatePresence mode="wait">
   374→            {service ? (
   375→              <motion.div
   376→                key={`${platform?.id}-${service.id}`}
   377→                initial={{ opacity: 0, x: 18 }}
   378→                animate={{ opacity: 1, x: 0 }}
   379→                exit={{ opacity: 0, x: -14 }}
   380→                transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   381→                className="space-y-5"
   382→              >
   383→                <div className="flex flex-wrap items-center gap-2">
   384→                  <h3 className="text-lg font-bold">
   385→                    <Tx text={service.name} />
   386→                  </h3>
   387→                  {service.warranty ? (
   388→                    <span className="rounded-full border border-emerald-500/40 bg-emerald-500/10 px-2 py-0.5 text-[10px] font-bold text-emerald-500">
   389→                      <Tx text={service.warranty} />
   390→                    </span>
   391→                  ) : null}
   392→                  {service.speed ? (
   393→                    <span className="flex items-center gap-1 rounded-full border border-cyan-500/40 bg-cyan-500/10 px-2 py-0.5 text-[10px] font-bold text-cyan-500">
   394→                      <Zap size={10} />
   395→                      <Tx text={service.speed} />
   396→                    </span>
   397→                  ) : null}
   398→                </div>
   399→
   400→                {service.description ? (
   401→                  <p className="text-sm leading-relaxed text-muted-foreground">
   402→                    <Tx text={service.description} />
   403→                  </p>
   404→                ) : null}
   405→
   406→                {/* quantity based */}
   407→                {!service.packages?.length && service.pricePerKUSD ? (
   408→                  <div className="space-y-4">
   409→                    <div className="flex items-center justify-between gap-3">
   410→                      <span className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   411→                        {t("shop.quantity")}
   412→                      </span>
   413→                      <div className="flex items-center gap-1.5">
   414→                        {[500, 1000, 5000, 10000].map((v) => (
   415→                          <button
   416→                            key={v}
   417→                            onClick={() => setQty(v)}
   418→                            className={cn(

   426→                      </div>
   427→                    </div>

   429→                    <input

   433→                      min={service.minQty ?? 100}

   450→                        aria-label={t("shop.quantity")}
   451→                      />
   452→                      <span className="text-xs text-muted-foreground">
   453→                        {t("shop.min")}: {service.minQty} · {t("shop.max")}: {(service.maxQty ?? 0).toLocaleString()} · {fmtUSD(service.pricePerKUSD ?? 0)} {t("shop.perK")}

   455→                    </div>
   456→
   457→                    {!isGame ? (

   500→                            </p>
   501→                          </motion.button>
   502→                        );
   503→                      })}
   504→                    </div>
   505→
   506→                    <div className="relative">
   507→                      <Gamepad2 size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   508→                      <TextInput
   509→                        id="cfg-player"
   510→                        value={playerId}
   511→                        onChange={(e) => setPlayerId(e.target.value)}
   512→                        placeholder={`${t("shop.uidLabel")} — ${t("shop.uidHint")}`}
   513→                        className="pl-9"
   514→                      />
   515→                    </div>
   516→                  </div>
   517→                ) : null}
   518→
   519→                {/* price + buy */}
   520→                <div className="flex flex-wrap items-center justify-between gap-4 rounded-2xl border border-acc-1/25 bg-gradient-to-r from-[color-mix(in_oklab,var(--acc-1)_10%,transparent)] to-[color-mix(in_oklab,var(--acc-2)_10%,transparent)] p-4">
   521→                  <div>
   522→                    <p className="text-[11px] font-semibold uppercase tracking-widest text-muted-foreground">
   523→                      {t("shop.price")}
   524→                    </p>
   525→                    <AnimatePresence mode="popLayout">
   526→                      <motion.p
   527→                        key={shown}
   528→                        initial={{ opacity: 0, y: 10 }}
   529→                        animate={{ opacity: 1, y: 0 }}
   530→                        exit={{ opacity: 0, y: -10 }}
   531→                        transition={{ duration: 0.22 }}
   532→                        className="font-[family-name:var(--f-display)] text-2xl font-extrabold text-gradient sm:text-3xl"
   533→                      >
   534→                        {shown}
   535→                      </motion.p>
   536→                    </AnimatePresence>
   537→                  </div>
   538→                  <AccentButton size="lg" onClick={buy} disabled={!price.valid || price.usd <= 0}>
   539→                    <ShoppingBag size={16} />
   540→                    {t("shop.buyNow")}
   541→                  </AccentButton>
   542→                </div>
   543→
   544→                {err ? (
   545→                  <motion.p initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="text-xs font-semibold text-red-400">
   546→                    {err}
   547→                  </motion.p>
   548→                ) : null}
   549→              </motion.div>
   550→            ) : (
   551→              <motion.div
   552→                key="empty"
   553→                initial={{ opacity: 0 }}
   554→                animate={{ opacity: 1 }}
   555→                className="flex h-64 items-center justify-center text-sm text-muted-foreground"
   556→              >
   557→                <span className="flex items-center gap-2">
   558→                  <PackageIcon size={16} />
   559→                  {t("shop.selectPrompt")}
   560→                </span>
   561→              </motion.div>
   562→            )}
   563→          </AnimatePresence>
   564→        </div>
   565→      </div>
   566→    </GlassCard>
   567→  );
   568→}
