   154→   154→export function CategoryBar({
   155→   155→  platforms,
   156→   156→  active,
   157→   157→  onSelect,
   158→   158→}: {
   159→   159→  platforms: Platform[];
   160→   160→  active: string | null;
   161→   161→  onSelect: (id: string) => void;
   162→   162→}) {
   163→   163→  const { t } = useI18n();
   164→   164→  const [kind, setKind] = useState<"social" | "game">("social");
   165→   165→
   166→   166→  const filtered = useMemo(() => platforms.filter((p) => p.kind === kind), [platforms, kind]);
   167→   167→
   168→   168→  // derived effective selection — no state-sync effects needed
   169→   169→  const activeId = active && filtered.some((p) => p.id === active) ? active : filtered[0]?.id ?? null;
   170→   170→
   171→   171→  const switchKind = (k: "social" | "game") => {
   172→   172→    setKind(k);
   173→   173→    const list = platforms.filter((p) => p.kind === k);
   174→   174→    if (list.length && !list.some((p) => p.id === active)) onSelect(list[0].id);
   175→   175→  };
   176→   176→
   177→   177→  return (
   178→   178→    <div className="space-y-4">
   179→   179→      <div className="flex justify-center">
   180→   180→        <Segmented
   181→   181→          value={kind}
   182→   182→          onChange={(v) => switchKind(v as "social" | "game")}
   183→   183→          options={[
   184→   184→            { value: "social", label: <span className="flex items-center gap-1.5"><Sparkles size={12} /> {t("shop.social")}</span> },
   185→   185→            { value: "game", label: <span className="flex items-center gap-1.5"><Gamepad2 size={12} /> {t("shop.games")}</span> },
   186→   186→          ]}
   187→   187→        />
   188→   188→      </div>
   189→   189→
   190→   190→      <div className="no-scrollbar -mx-4 flex gap-2 overflow-x-auto px-4 pb-1 sm:mx-0 sm:flex-wrap sm:justify-center sm:px-0">
   191→   191→        <AnimatePresence mode="popLayout">
   192→   192→          {filtered.map((p, i) => {
   193→   193→            const isActive = p.id === activeId;
   194→   194→            return (
   195→   195→              <motion.button
   196→   196→                layout
   197→   197→                key={p.id}
   198→   198→                initial={{ opacity: 0, scale: 0.85, y: 10 }}
   199→   199→                animate={{ opacity: 1, scale: 1, y: 0 }}
   200→   200→                exit={{ opacity: 0, scale: 0.9 }}
   201→   201→                transition={{ delay: i * 0.03, type: "spring", stiffness: 380, damping: 28 }}
   202→   202→                whileHover={{ y: -2 }}
   203→   203→                whileTap={{ scale: 0.95 }}
   204→   204→                onClick={() => onSelect(p.id)}
   205→   205→                aria-pressed={isActive}
   206→   206→                className={cn(
   207→   207→                  "glass flex shrink-0 items-center gap-2 rounded-full px-4 py-2.5 text-sm font-semibold transition-all",
   208→   208→                  isActive && "pill-active",
   209→   209→                )}
   210→   210→                style={isActive ? { color: p.color } : undefined}
   211→   211→              >
   212→   212→                <PlatformIcon id={p.id} size={17} className={cn(!isActive && "text-muted-foreground")} />
   213→   213→                <span className={cn(!isActive && "text-muted-foreground")}>{p.name}</span>
   214→   214→              </motion.button>
   215→   215→            );
   216→   216→          })}
   217→   217→        </AnimatePresence>
   218→   218→      </div>
   219→   219→    </div>
   220→   220→  );
   221→   221→}
