   153→
   154→export function CategoryBar({
   155→  platforms,
   156→  active,
   157→  onSelect,
   158→}: {
   159→  platforms: Platform[];
   160→  active: string | null;
   161→  onSelect: (id: string) => void;
   162→}) {
   163→  const { t } = useI18n();
   164→  const [kind, setKind] = useState<"social" | "game">("social");
   165→
   166→  const filtered = useMemo(() => platforms.filter((p) => p.kind === kind), [platforms, kind]);
   167→
   168→  // derived effective selection — no state-sync effects needed
   169→  const activeId = active && filtered.some((p) => p.id === active) ? active : filtered[0]?.id ?? null;
   170→
   171→  const switchKind = (k: "social" | "game") => {
   172→    setKind(k);
   173→    const list = platforms.filter((p) => p.kind === k);
   174→    if (list.length && !list.some((p) => p.id === active)) onSelect(list[0].id);
   175→  };
   176→
   177→  return (
   178→    <div className="space-y-4">
   179→      <div className="flex justify-center">
   180→        <Segmented
   181→          value={kind}
   182→          onChange={(v) => switchKind(v as "social" | "game")}
   183→          options={[
   184→            { value: "social", label: <span className="flex items-center gap-1.5"><Sparkles size={12} /> {t("shop.social")}</span> },
   185→            { value: "game", label: <span className="flex items-center gap-1.5"><Gamepad2 size={12} /> {t("shop.games")}</span> },
   186→          ]}
   187→        />
   188→      </div>
   189→
   190→      <div className="no-scrollbar -mx-4 flex gap-2 overflow-x-auto px-4 pb-1 sm:mx-0 sm:flex-wrap sm:justify-center sm:px-0">
   191→        <AnimatePresence mode="popLayout">
   192→          {filtered.map((p, i) => {
   193→            const isActive = p.id === activeId;
   194→            return (
   195→              <motion.button
   196→                layout
   197→                key={p.id}
   198→                initial={{ opacity: 0, scale: 0.85, y: 10 }}
   199→                animate={{ opacity: 1, scale: 1, y: 0 }}
   200→                exit={{ opacity: 0, scale: 0.9 }}
   201→                transition={{ delay: i * 0.03, type: "spring", stiffness: 380, damping: 28 }}
   202→                whileHover={{ y: -2 }}
   203→                whileTap={{ scale: 0.95 }}
   204→                onClick={() => onSelect(p.id)}
   205→                aria-pressed={isActive}
   206→                className={cn(
   207→                  "glass flex shrink-0 items-center gap-2 rounded-full px-4 py-2.5 text-sm font-semibold transition-all",
   208→                  isActive && "pill-active",
   209→                )}
   210→                style={isActive ? { color: p.color } : undefined}
   211→              >
   212→                <PlatformIcon id={p.id} size={17} className={cn(!isActive && "text-muted-foreground")} />
   213→                <span className={cn(!isActive && "text-muted-foreground")}>{p.name}</span>
   214→              </motion.button>
   215→            );
   216→          })}
   217→        </AnimatePresence>
   218→      </div>
   219→    </div>
   220→  );
   221→}

   260→export function Configurator({
   261→  platformId,
   262→  onSelectPlatform,
   263→}: {
   264→  platformId: string | null;
   265→  onSelectPlatform: (id: string) => void;
   266→}) {
   267→  const { t } = useI18n();
   268→  const { platforms } = useCatalog();
   269→  const { settings } = useSettings();
   270→  const { currency } = useUI();
   271→  const draft = useDraft();
   272→
   273→  const [typeId, setTypeId] = useState<string | null>(draft.typeId);
   274→  const [qty, setQty] = useState<number>(draft.quantity ?? 1000);
   275→  const [packageId, setPackageId] = useState<string | null>(draft.packageId);
   276→  const [playerId, setPlayerId] = useState(draft.playerId ?? "");
   277→  const [link, setLink] = useState(draft.link ?? "");
   278→  const [err, setErr] = useState("");
   279→
   280→  /* derived selection chain — zero sync effects */
   281→  const platform = platforms.find((p) => p.id === platformId) ?? platforms[0] ?? null;
   282→  const activeServices = useMemo(
   283→    () => platform?.services.filter((s) => s.active) ?? [],
   284→    [platform],
   285→  );
   286→  const service = activeServices.find((s) => s.id === typeId) ?? activeServices[0] ?? null;
   287→  const isGame = platform?.kind === "game";
   288→  const effPackageId =
   289→    service?.packages?.length
   290→      ? service.packages.some((p) => p.id === packageId)
   291→        ? packageId
   292→        : service.packages[0].id
   293→      : null;
   294→
   295→  const price = useMemo(() => {
   296→    if (!service) return { usd: 0, valid: false, qty: 0 };
   297→    if (service.packages?.length) {
   298→      const pk = findPackage(service, effPackageId);
   299→      return { usd: pk?.priceUSD ?? 0, valid: true, qty: pk?.qty ?? 0 };
   300→    }
   301→    const r = calcQtyPrice(service, qty);
   302→    return { usd: r.usd, valid: r.valid, qty: r.qty };
   303→  }, [service, effPackageId, qty]);
   304→
   305→  const shown = currency === "NPR" ? fmtNPR(convertUSD(price.usd, settings.exchangeRate)) : fmtUSD(price.usd);
   306→
   307→  const buy = () => {
   308→    setErr("");
   309→    if (!platform || !service) return;
   310→    if (service.packages?.length && !effPackageId) {
   311→      setErr(t("shop.package"));
   312→      return;
   313→    }
   314→    if (!service.packages?.length && !price.valid) {
   315→      setErr(`${t("shop.min")}: ${service.minQty ?? 100}`);
   316→      return;
   317→    }
   318→    if (isGame && !playerId.trim()) {
   319→      setErr(t("shop.uidHint"));
   320→      document.getElementById("cfg-player")?.focus();
   321→      return;
   322→    }
   323→    draft.setDraft({
   324→      platformId: platform.id,
   325→      typeId: service.id,
   326→      quantity: service.packages?.length ? null : price.qty,
   327→      packageId: service.packages?.length ? effPackageId : null,
   328→      playerId: playerId.trim(),
   329→      link: link.trim(),
   330→    });
   331→    navigate("/checkout");
   332→  };
   333→
   334→  const sliderFill =
   335→    service && service.maxQty && service.maxQty > (service.minQty ?? 100)
   336→      ? `${(((qty - (service.minQty ?? 100)) / (service.maxQty - (service.minQty ?? 100))) * 100).toFixed(1)}%`
   337→      : "50%";
   338→
   339→  return (
   340→    <GlassCard className="overflow-hidden p-0">
   341→      <div className="grid lg:grid-cols-[320px_1fr]">
   342→        {/* service list */}
   343→        <div className="border-b border-border/60 p-5 lg:border-b-0 lg:border-r">
   344→          {platform ? (
   345→            <div className="mb-4 flex items-center gap-3">
   346→              <span className="glass flex h-11 w-11 items-center justify-center rounded-xl" style={{ color: platform.color }}>
   347→                <PlatformIcon id={platform.id} size={22} />
   348→              </span>
   349→              <div className="min-w-0">
   350→                <p className="truncate text-sm font-bold">{platform.name}</p>
   351→                <p className="text-[11px] text-muted-foreground">
   352→                  {activeServices.length} {t("nav.services").toLowerCase()}
   353→                </p>
   354→              </div>
   355→            </div>
   356→          ) : null}
   357→
   358→          <div className="max-h-72 space-y-1.5 overflow-y-auto pr-1 lg:max-h-[420px]">
   359→            {activeServices.map((s) => (
   360→              <ServiceRow key={s.id} service={s} active={s.id === service?.id} onClick={() => setTypeId(s.id)} />
   361→            ))}
   362→            {platform && activeServices.length === 0 ? (
   363→              <p className="py-8 text-center text-xs text-muted-foreground">{t("shop.selectPrompt")}</p>
   364→            ) : null}
   365→          </div>
   366→        </div>
   367→
   368→        {/* config panel */}
   369→        <div className="p-5 sm:p-6">
   370→          <AnimatePresence mode="wait">
   371→            {service ? (
   372→              <motion.div
   373→                key={`${platform?.id}-${service.id}`}
   374→                initial={{ opacity: 0, x: 18 }}
   375→                animate={{ opacity: 1, x: 0 }}
   376→                exit={{ opacity: 0, x: -14 }}
   377→                transition={{ duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   378→                className="space-y-5"
   379→              >
   380→                <div className="flex flex-wrap items-center gap-2">
   381→                  <h3 className="text-lg font-bold">
   382→                    <Tx text={service.name} />
   383→                  </h3>
   384→                  {service.warranty ? (
   385→                    <span className="rounded-full border border-emerald-500/40 bg-emerald-500/10 px-2 py-0.5 text-[10px] font-bold text-emerald-500">
   386→                      <Tx text={service.warranty} />
   387→                    </span>
   388→                  ) : null}
   389→                  {service.speed ? (
   390→                    <span className="flex items-center gap-1 rounded-full border border-cyan-500/40 bg-cyan-500/10 px-2 py-0.5 text-[10px] font-bold text-cyan-500">
   391→                      <Zap size={10} />
   392→                      <Tx text={service.speed} />
   393→                    </span>
   394→                  ) : null}
   395→                </div>
   396→
   397→                {service.description ? (
   398→                  <p className="text-sm leading-relaxed text-muted-foreground">
   399→                    <Tx text={service.description} />
   400→                  </p>
   401→                ) : null}
   402→
   403→                {/* quantity based */}
   404→                {!service.packages?.length && service.pricePerKUSD ? (
   405→                  <div className="space-y-4">
   406→                    <div className="flex items-center justify-between gap-3">
   407→                      <span className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   408→                        {t("shop.quantity")}
   409→                      </span>
   410→                      <div className="flex items-center gap-1.5">
   411→                        {[500, 1000, 5000, 10000].map((v) => (
   412→                          <button
   413→                            key={v}
   414→                            onClick={() => setQty(v)}
   415→                            className={cn(

   423→                      </div>
   424→                    </div>

   426→                    <input

   430→                      min={service.minQty ?? 100}

   450→                        {t("shop.min")}: {service.minQty} · {t("shop.max")}: {(service.maxQty ?? 0).toLocaleString()} · {fmtUSD(service.pricePerKUSD ?? 0)} {t("shop.perK")}

   454→                    {!isGame ? (

   516→                {/* mandatory drop disclaimer */}

   526→

   528→                <div className="sticky bottom-4 z-20 flex flex-wrap items-center justify-between gap-4 rounded-2xl border border-acc-1/25 bg-gradient-to-r from-[color-mix(in_oklab,var(--acc-1)_10%,transparent)] to-[color-mix(in_oklab,var(--acc-2)_10%,transparent)] p-4 shadow-[0_18px_50px_-12px_rgba(0,0,0,0.5)] backdrop-blur-xl">

   544→                    </AnimatePresence>

   550→                </div>

   552→                {err ? (
   553→                  <motion.p initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="text-xs font-semibold text-red-400">
   554→                    {err}
   555→                  </motion.p>

   557→              </motion.div>
   558→            ) : (
   559→              <motion.div
   560→                key="empty"
   561→                initial={{ opacity: 0 }}

   566→                  <PackageIcon size={16} />
   567→                  {t("shop.selectPrompt")}
   568→                </span>
   569→              </motion.div>
   570→            )}
   571→          </AnimatePresence>
   572→        </div>
   573→      </div>
   574→    </GlassCard>
   575→  );
   576→}
