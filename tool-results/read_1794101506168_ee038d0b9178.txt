    73→
    74→function Gate({ onPass }: { onPass: () => void }) {
    75→  const { t } = useI18n();
    76→  const { settings } = useSettings();
    77→  const [username, setUsername] = useState("");
    78→  const [password, setPassword] = useState("");
    79→  const [wrong, setWrong] = useState(false);
    80→  const [busy, setBusy] = useState(false);
    81→
    82→  const submit = async () => {
    83→    if (!username.trim() || !password) return;
    84→    setBusy(true);
    85→    const [userHash, passHash] = await Promise.all([
    86→      sha256Hex(username.trim().toLowerCase()),
    87→      sha256Hex(password),
    88→    ]);
    89→    const ok =
    90→      userHash === (settings.adminUserHash || DEFAULT_ADMIN_USER_HASH) &&
    91→      passHash === (settings.adminPassHash || DEFAULT_ADMIN_PASS_HASH);
    92→    setBusy(false);
    93→    if (ok) {
    94→      try {
    95→        sessionStorage.setItem(LS_ADMIN_OK, "1");
    96→      } catch {
    97→        /* ignore */
    98→      }
    99→      elevateLocalAdmin();
   100→      onPass();
   101→    } else {
   102→      setWrong(true);
   103→      setTimeout(() => setWrong(false), 2200);
   104→    }
   105→  };
   106→
   107→  return (
   108→    <section className="mx-auto flex min-h-[70vh] max-w-md items-center px-4 pt-24">
   109→      <motion.div initial={{ opacity: 0, y: 30 }} animate={{ opacity: 1, y: 0 }} className="w-full">
   110→        <GlassCard strong className="p-8 text-center">
   111→          <motion.div
   112→            animate={wrong ? { x: [0, -10, 10, -6, 6, 0] } : {}}
   113→            transition={{ duration: 0.4 }}
   114→            className="mx-auto mb-5 flex h-16 w-16 items-center justify-center rounded-2xl"
   115→            style={{ background: "linear-gradient(135deg, color-mix(in oklab, var(--acc-1) 22%, transparent), color-mix(in oklab, var(--acc-2) 22%, transparent))" }}
   116→          >
   117→            <Lock size={26} className={cn(wrong ? "text-red-400" : "text-acc-1")} />
   118→          </motion.div>
   119→          <h1 className="text-xl font-extrabold">{t("admin.gate")}</h1>
   120→          <p className="mx-auto mt-1.5 max-w-xs text-xs leading-relaxed text-muted-foreground">{t("admin.gateSub")}</p>
   121→
   122→          <form
   123→            className="mt-6 space-y-3"
   124→            onSubmit={(e) => {
   125→              e.preventDefault();
   126→              void submit();
   127→            }}
   128→          >
   129→            <TextInput
   130→              value={username}
   131→              onChange={(e) => setUsername(e.target.value)}
   132→              placeholder={t("admin.username")}
   133→              className={cn("text-center font-semibold", wrong && "!border-red-500")}
   134→              autoFocus
   135→              autoComplete="username"
   136→            />
   137→            <TextInput
   138→              type="password"
   139→              value={password}
   140→              onChange={(e) => setPassword(e.target.value)}
   141→              placeholder={t("admin.password")}
   142→              className={cn("text-center font-mono tracking-[0.3em]", wrong && "!border-red-500")}
   143→              autoComplete="current-password"
   144→            />
   145→            {wrong ? <p className="text-xs font-bold text-red-400">{t("admin.wrong")}</p> : null}
   146→            <AccentButton type="submit" className="w-full" size="lg" disabled={busy}>
   147→              <KeyRound size={16} />
   148→              {t("admin.enter")}
   149→            </AccentButton>
   150→          </form>
   151→        </GlassCard>
   152→      </motion.div>
   153→    </section>
   154→  );
   155→}
